  ```bash
  # 腳本全文（本輪實跑於暫存目錄；接手者可原文另存為 coverage_check.sh 後 bash 執行，不需任何額外相依）
  #!/usr/bin/env bash
  cd /c/Users/excal/Projects/Claude/soft-dev-team-260919 || exit 1
  SD=docs/specs/03_系統設計書_SD.md
  TRC=docs/specs/traceability.md

  # 1) 從 SD §3.1~3.3 抽出「模組 -> 對應UC」原文
  sed -n '/^### 3.1 後端模組/,/^### 3.4/p' "$SD" \
  | awk -F'|' '/^\| \*\*(BE|FE|OPS)-/ {
      m=$2; uc=$4;
      gsub(/\*\*/,"",m); gsub(/`[^`]*`/,"",m); gsub(/\(P1\)/,"",m);
      gsub(/^ +| +$/,"",m); gsub(/^ +| +$/,"",uc);
      print m "\t" uc
    }' > /tmp/mod_uc_raw.txt

  # 2) 把 UC-00a~UC-00b 範圍寫法展開；「全部…」展開為 ALL
  expand_uc() {
    local s="$1" out=""
    while [[ "$s" =~ UC-([0-9]{3})~UC-([0-9]{3}) ]]; do
      local a=${BASH_REMATCH[1]} b=${BASH_REMATCH[2]} i
      for ((i=10#$a; i<=10#$b; i++)); do out="$out UC-$(printf %03d $i)"; done
      s=${s/UC-$a~UC-$b/}
    done
    for u in $(echo "$s" | grep -oE 'UC-[0-9]{3}'); do out="$out $u"; done
    if [[ "$s" == *"全部"* ]]; then out="$out ALL"; fi
    echo "$out"
  }
  declare -A MODUC
  while IFS=$'\t' read -r m uc; do MODUC["$m"]=$(expand_uc "$uc"); done < /tmp/mod_uc_raw.txt
  echo "=== 模組 -> 展開後對應 UC ==="
  for m in $(printf '%s\n' "${!MODUC[@]}" | sort); do printf '%-8s %s\n' "$m" "${MODUC[$m]}"; done

  # 3) 逐 US 反查：其 UC 未被任一所列模組明載（且該模組非 ALL）即計為缺口
  awk -F'|' '/^\| US-0/ && NF>=8 {
    us=$2; uc=$5; md=$6;
    gsub(/^ +| +$/,"",us); gsub(/^ +| +$/,"",uc); gsub(/^ +| +$/,"",md); gsub(/`/,"",md);
    print us "\t" uc "\t" md
  }' "$TRC" > /tmp/us_rows.txt
  echo; echo "=== 逐 US 覆蓋度判定 ==="
  GAP=0
  while IFS=$'\t' read -r us ucs mds; do
    uclist=$(echo "$ucs" | grep -oE 'UC-[0-9]{3}')
    modlist=$(echo "$mds" | grep -oE '(BE|FE|OPS)-[0-9]{2}')
    hit=""; miss=""
    for m in $modlist; do
      ok=0
      for u in $uclist; do
        if [[ " ${MODUC[$m]} " == *" $u "* || " ${MODUC[$m]} " == *" ALL "* ]]; then ok=1; fi
      done
      if [[ $ok -eq 1 ]]; then hit="$hit $m"; else miss="$miss $m"; fi
    done
    if [[ -z "$hit" ]]; then verdict="缺口"; GAP=$((GAP+1)); else verdict="有對應"; fi
    printf '%-8s %-16s %-8s SD明載=%s%s\n' "$us" "$ucs" "$verdict" "${hit:- 無}" \
      "${miss:+ ；TRC較寬(SD對應UC欄未明列)=$miss}"
  done < /tmp/us_rows.txt
  echo; echo "缺口總數（無任何 SD 模組明載其 UC 的 US）：$GAP"
  ```
